func canBeValid(s string, locked string) bool {
    if len(s) % 2 == 1 { 
        return false
    }
    open, close := 0, 0
    for i := 0; i < len(s); i++ {
        if s[i] == '(' || locked[i] == '0' {
            open++
        } else {
            open--
        }
        if open < 0 { 
            return false 
        }
    }
    if open == 0 {
        return true
    }
    for i := len(s) - 1; i >= 0; i-- {
        if s[i] == ')' || locked[i] == '0' {
            close++
        } else {
            close--
        }
        if close < 0 {
            return false 
        }
    }
    return true
}

# Complexity :
- Time : O(N)
- Space : O(1)
