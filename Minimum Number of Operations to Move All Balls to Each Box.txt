class Solution {
    public int[] minOperations(String boxes) {
        int n = boxes.length();
        int[] res = new int[n];
        for(int i = 0; i < n; i++){
            int count = 0;
            for(int j = 0; j < n; j++){
                if(boxes.charAt(j) == '1' && j != i){
                    count += Math.abs(j - i);
                    res[i] = count;
                }
            }
        }
        return res;
    }
}

# Complexity :
- Time : O(N ^ 2)
- Space : O(N)
