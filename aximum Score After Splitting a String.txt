class Solution {
    public int maxScore(String s) {
        int res = 0, n = s.length(), ones = 0, zeros = 0;
        for(int i = 0; i < n; i++){
            if(s.charAt(i) == '1'){
                ones++;
            }
        }
        for(int i = 0; i < n - 1; i++){
            if(s.charAt(i) == '0'){
                zeros++;
            }else{
                ones--;
            }
            res = Math.max(res, zeros + ones);
        }
        return res;
    }
}

# Complexity :
- Time : O(N)
- Space : O(1)
